Try it now:
curl -X POST https://justin-advisors-thirty-recommend.trycloudflare.com/v1/extract \
  -H 'Content-Type: application/json' \
  -d '{
    "text": "From: sarah.johnson@company.com\nSubject: Q4 Meeting\nHi team, lets meet on Dec 15th at 2pm in conference room B.\nThanks,\nSarah Johnson\nSenior Manager\n(555) 123-4567",
    "schema": {
      "type": "object",
      "properties": {
        "sender_email": {"type": "string"},
        "sender_name": {"type": "string"},
        "meeting_date": {"type": "string"},
        "meeting_time": {"type": "string"}
      }
    }
  }'
